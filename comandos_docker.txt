docker --version — Muestra la versión de Docker
docker compose version — Muestra la versión de Docker Compose
docker pull <imagen> — Descarga una imagen desde Docker Hub
docker run --rm -v "<host>:/app" -w /app composer:2 create-project laravel/laravel . --prefer-dist --no-interaction — Scaffold de Laravel con Composer (contenedor temporal)
docker run --rm -v "<host>:/app" -w /app node:20-alpine sh -c "npm install --no-audit --strict-ssl false" — Instala dependencias Node dentro de un contenedor
docker run --rm -v "<host>:/app" -w /app node:20-alpine npm run build — Compila assets con Vite en contenedor
docker compose build — Construye las imágenes desde el Dockerfile
docker compose build --no-cache — Reconstruye imágenes sin usar cache
docker compose up -d — Levanta los servicios en segundo plano
docker compose logs -f app — Sigue los logs del servicio `app`
docker compose down — Detiene y elimina contenedores
docker compose down -v — Detiene contenedores y elimina volúmenes
docker exec -it sae_dashboard sh — Abre shell interactivo en el contenedor `sae_dashboard`
docker exec -it sae_dashboard php artisan <comando> — Ejecuta un comando Artisan dentro del contenedor
docker run --rm -v "<host>:/app" -w /app php:8.3-alpine sh -c "php artisan key:generate --force" — Ejecuta Artisan en contenedor temporal (ej.: generar APP_KEY)

# Comandos para detener / arrancar contenedores
docker compose stop                 — Detiene los contenedores (no los elimina)
docker compose start                — Inicia contenedores ya creados
docker compose restart              — Reinicia los servicios (stop + start)
docker compose down                 — Detiene y elimina contenedores (y red)
docker compose down -v              — Detiene, elimina contenedores y volúmenes

# Comandos por contenedor individual
docker stop <container>             — Detiene un contenedor por nombre/ID
docker start <container>            — Inicia un contenedor detenido por nombre/ID
docker restart <container>          — Reinicia un contenedor específico

# Ver estado / gestión
docker compose ps                   — Lista contenedores del composition actual
docker ps -a                        — Lista todos los contenedores (incl. parados)
docker compose logs -f <service>    — Seguir logs de un servicio (ej: sae_nlp)

# Reconstruir y levantar (útil al cambiar Dockerfile o dependencias)
docker compose build --no-cache nlp
docker compose up -d --no-deps --build nlp
